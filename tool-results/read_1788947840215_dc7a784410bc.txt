   152→  { label: 'Add Student', hint: 'Students', module: 'students', icon: UserPlus },
   153→  { label: 'Add Teacher', hint: 'Staff', module: 'teachers', icon: Briefcase },
   154→  { label: 'New Admission Lead', hint: 'CRM', module: 'admissions', icon: UserPlus },

   162→
   163→function CommandPalette() {
   164→  const [open, setOpen] = React.useState(false)
   165→  const [q, setQ] = React.useState('')
   166→  const [results, setResults] = React.useState<any[]>([])
   167→  const [loading, setLoading] = React.useState(false)
   168→  const [activeIdx, setActiveIdx] = React.useState(0)
   169→  const { setModule } = useErp()
   170→
   171→  React.useEffect(() => {
   172→    if (!open) { setQ(''); setResults([]); setActiveIdx(0) }
   173→  }, [open])
   174→
   175→  React.useEffect(() => {
   176→    if (q.trim().length < 2) { setResults([]); setLoading(false); return }
   177→    setLoading(true)
   178→    const t = setTimeout(() => {
   179→      api<{ data: any[] }>(`/api/search?q=${encodeURIComponent(q.trim())}`)
   180→        .then(res => { setResults(res.data || []); setActiveIdx(0) })
   181→        .catch(() => setResults([]))
   182→        .finally(() => setLoading(false))
   183→    }, 280)
   184→    return () => clearTimeout(t)
   185→  }, [q])
   186→
   187→  React.useEffect(() => {
   188→    const onKey = (e: KeyboardEvent) => {
   189→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); setOpen(o => !o) }
   190→      if (e.key === 'Escape') setOpen(false)
   191→    }
   192→    window.addEventListener('keydown', onKey)
   193→    return () => window.removeEventListener('keydown', onKey)
   194→  }, [])
   195→
   196→  const flat = React.useMemo(() => {
   197→    // Order results into a flat list mirroring the grouped render
   198→    const groups: Record<string, any[]> = {}
   199→    results.forEach(r => { (groups[r.type] ||= []).push(r) })
   200→    return Object.entries(groups).flatMap(([, arr]) => arr)
   201→  }, [results])
   202→
   203→  const goModule = (m: string) => { setModule(m); setOpen(false) }
   204→  const goResult = (r: any) => { setModule(r.module); setOpen(false) }
   205→
   206→  const onKeyDown = (e: React.KeyboardEvent) => {
   207→    const list = q.trim().length >= 2 ? flat : []
   208→    if (e.key === 'ArrowDown') { e.preventDefault(); setActiveIdx(i => (i + 1) % Math.max(1, list.length)) }
   209→    else if (e.key === 'ArrowUp') { e.preventDefault(); setActiveIdx(i => (i - 1 + list.length) % Math.max(1, list.length)) }
   210→    else if (e.key === 'Enter') {
   211→      e.preventDefault()
   212→      if (list[activeIdx]) goResult(list[activeIdx])
   213→    }
   214→  }
   215→
   216→  const grouped = React.useMemo(() => {
   217→    const groups: Record<string, any[]> = {}
   218→    results.forEach(r => { (groups[r.type] ||= []).push(r) })
   219→    return Object.entries(groups)
   220→  }, [results])
   221→
   222→  return (
   223→    <div className="relative w-full max-w-md">
   224→      <button
   225→        onClick={() => setOpen(true)}
   226→        aria-label="Open global search"
   227→        className="flex h-9 w-full items-center gap-2 rounded-xl border bg-muted/40 px-3 text-sm text-muted-foreground transition-all duration-200 hover:border-primary/30 hover:bg-muted hover:shadow-sm md:w-64 lg:w-80">
   228→        <Search className="h-4 w-4 shrink-0 transition-transform duration-200" />
   229→        <span className="flex-1 truncate text-left">Search or jump to…</span>
   230→        <kbd className="hidden rounded-md border bg-background px-1.5 py-0.5 font-mono text-[10px] font-medium md:inline-block">⌘K</kbd>
   231→      </button>
   232→
   233→      {open && (
   234→        <div className="anim-fade-in fixed inset-0 z-50 bg-black/45 backdrop-blur-[2px]" onClick={() => setOpen(false)}>
   235→          <div
   236→            className="anim-rise mx-auto mt-[10vh] w-[92%] max-w-xl overflow-hidden rounded-2xl border bg-popover shadow-[var(--shadow-pop)]"
   237→            onClick={e => e.stopPropagation()}>
   238→            {/* Input */}
   239→            <div className="flex items-center gap-2.5 border-b px-4">
   240→              <Search className="h-4 w-4 shrink-0 text-muted-foreground" />
   241→              <Input autoFocus value={q} onChange={e => setQ(e.target.value)} onKeyDown={onKeyDown}
   242→                placeholder="Search students, staff, fees, books…"
   243→                className="h-13 border-0 bg-transparent py-4 shadow-none focus-visible:ring-0" />
   244→              {q && <button onClick={() => setQ('')} aria-label="Clear search"><X className="h-4 w-4 text-muted-foreground transition-colors hover:text-foreground" /></button>}
   245→              <kbd className="hidden rounded-md border bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground sm:block">ESC</kbd>
   246→            </div>
   247→
   248→            <div className="max-h-[56vh] overflow-y-auto p-2">
   249→              {/* Quick actions (empty query) */}
   250→              {q.trim().length < 2 && (
   251→                <>
   252→                  <p className="px-2 pb-1.5 pt-1.5 text-[10px] font-semibold uppercase tracking-[0.14em] text-muted-foreground/70">Quick actions</p>
   253→                  {QUICK_ACTIONS.map((a, i) => (
   254→                    <button key={a.label} onClick={() => goModule(a.module)}
   255→                      className="group flex w-full items-center gap-3 rounded-lg px-2.5 py-2.5 text-left transition-all duration-150 hover:bg-muted"
   256→                      style={{ ['--delay' as string]: `${i * 22}ms` }}>
   257→                      <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-primary/10 text-primary transition-transform duration-150 group-hover:scale-105">
   258→                        <a.icon className="h-4 w-4" />
   259→                      </span>
   260→                      <span className="min-w-0 flex-1">
   261→                        <span className="block truncate text-sm font-medium">{a.label}</span>
   262→                        <span className="block truncate text-xs text-muted-foreground">{a.hint}</span>
   263→                      </span>
   264→                      <CornerDownLeft className="h-3.5 w-3.5 text-muted-foreground opacity-0 transition-opacity group-hover:opacity-60" />
   265→                    </button>
   266→                  ))}
   267→                </>
   268→              )}
   269→
   270→              {/* Search results */}
   271→              {q.trim().length >= 2 && (
   272→                <>
   273→                  {loading && <div className="space-y-1.5 p-2">{Array.from({ length: 3 }).map((_, i) => <div key={i} className="skeleton-shimmer h-11 rounded-lg" />)}</div>}
   274→                  {!loading && grouped.length === 0 && (
   275→                    <div className="px-4 py-10 text-center">
   276→                      <Search className="mx-auto mb-2 h-6 w-6 text-muted-foreground/50" />
   277→                      <p className="text-sm font-medium">No results for “{q}”</p>
   278→                      <p className="mt-1 text-xs text-muted-foreground">Try a name, ID, phone, email or admission number.</p>
   279→                    </div>
   280→                  )}
   281→                  {!loading && grouped.map(([type, arr]) => {
   282→                    const GIcon = RESULT_ICONS[type] || Search
   283→                    return (
   284→                      <div key={type} className="pb-1">
   285→                        <p className="flex items-center gap-1.5 px-2 pb-1 pt-2.5 text-[10px] font-semibold uppercase tracking-[0.14em] text-muted-foreground/70">
   286→                          <GIcon className="h-3 w-3" /> {type.charAt(0) + type.slice(1).toLowerCase()}s
   287→                        </p>
   288→                        {arr.map(r => {
   289→                          const idx = flat.indexOf(r)
   290→                          const Icon = RESULT_ICONS[type] || Search
   291→                          return (
   292→                            <button key={`${r.type}-${r.id}`} onClick={() => goResult(r)} onMouseEnter={() => setActiveIdx(idx)}
   293→                              className={cn('group flex w-full items-center gap-3 rounded-lg px-2.5 py-2.5 text-left transition-colors duration-150',
   294→                                idx === activeIdx && 'bg-primary/8 dark:bg-primary/15')}>
   295→                              <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-muted text-muted-foreground transition-colors group-hover:bg-primary/10 group-hover:text-primary">
   296→                                <Icon className="h-4 w-4" />
   297→                              </span>
   298→                              <span className="min-w-0 flex-1">
   299→                                <span className="block truncate text-sm font-medium">{r.title}</span>
   300→                                <span className="block truncate text-xs text-muted-foreground">{r.subtitle}</span>
   301→                              </span>
   302→                              {idx === activeIdx && <CornerDownLeft className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />}
   303→                            </button>
   304→                          )
   305→                        })}
   306→                      </div>
   307→                    )
   308→                  })}
   309→                </>
   310→              )}
   311→            </div>
   312→
   313→            {/* Footer hints */}
   314→            <div className="flex items-center gap-4 border-t bg-muted/30 px-4 py-2 text-[10.5px] text-muted-foreground">
   315→              <span className="flex items-center gap-1"><ArrowUp className="h-3 w-3" /><ArrowDown className="h-3 w-3" /> navigate</span>
   316→              <span className="flex items-center gap-1"><CornerDownLeft className="h-3 w-3" /> open</span>
   317→              <span className="ml-auto flex items-center gap-1"><Sparkles className="h-3 w-3 text-primary" /> Powered by Greenwood Search</span>
   318→            </div>
   319→          </div>
   320→        </div>
   321→      )}
   322→    </div>
   323→  )

   331→
   332→function NotificationBell() {
   333→  const [items, setItems] = React.useState<any[]>([])
   334→  const [unread, setUnread] = React.useState(0)
   335→  const [open, setOpen] = React.useState(false)
   336→  const { setModule } = useErp()
   337→
   338→  const load = React.useCallback(() => {
   339→    api<{ data: any[]; unread: number }>('/api/notifications')
   340→      .then(res => { setItems(res.data); setUnread(res.unread) })
   341→      .catch(() => {})
   342→  }, [])
   343→
   344→  React.useEffect(() => { load() }, [load])
   345→  React.useEffect(() => {
   346→    if (!open) return
   347→    const t = setInterval(load, 15000)
   348→    return () => clearInterval(t)
   349→  }, [open, load])
   350→
   351→  const markAll = async () => {
   352→    try { await api('/api/notifications', { method: 'PATCH', body: JSON.stringify({ all: true }) }); load() } catch { }
   353→  }
   354→
   355→  const markOne = async (n: any) => {
   356→    try { await api('/api/notifications', { method: 'PATCH', body: JSON.stringify({ id: n.id }) }); load() } catch { }
   357→  }
   358→
   359→  const toneCls: Record<string, string> = {
   360→    INFO: 'bg-sky-500/12 text-sky-600 dark:text-sky-400',
   361→    SUCCESS: 'bg-emerald-500/12 text-emerald-600 dark:text-emerald-400',
   362→    WARNING: 'bg-amber-500/12 text-amber-600 dark:text-amber-400',
   363→    DANGER: 'bg-rose-500/12 text-rose-600 dark:text-rose-400',
   364→    ALERT: 'bg-rose-500/12 text-rose-600 dark:text-rose-400',
   365→  }
   366→
   367→  return (
   368→    <DropdownMenu open={open} onOpenChange={setOpen}>
   369→      <DropdownMenuTrigger asChild>
   370→        <Button variant="outline" size="icon" className="relative h-9 w-9 rounded-xl transition-all hover:shadow-sm" aria-label={`Notifications${unread ? ` (${unread} unread)` : ''}`}>
   371→          <Bell className="h-4 w-4" />
   372→          {unread > 0 && (
   373→            <span className="anim-pop absolute -right-1 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 px-1 text-[10px] font-bold text-white shadow-sm">
   374→              {unread > 9 ? '9+' : unread}
   375→            </span>
   376→          )}
   377→        </Button>
   378→      </DropdownMenuTrigger>
   379→      <DropdownMenuContent align="end" className="w-80 rounded-xl p-0 sm:w-96">
   380→        <div className="flex items-center justify-between px-4 py-3">
   381→          <p className="font-display text-sm font-bold">Notifications</p>
   382→          {unread > 0 && <button onClick={markAll} className="text-xs font-semibold text-primary transition-colors hover:underline">Mark all read</button>}
   383→        </div>
   384→        <DropdownMenuSeparator className="my-0" />
   385→        <div className="max-h-[60vh] overflow-y-auto">
   386→          {items.length === 0 && (
   387→            <div className="px-4 py-10 text-center">
   388→              <span className="mx-auto mb-2 flex h-10 w-10 items-center justify-center rounded-full bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"><BadgeCheck className="h-5 w-5" /></span>
   389→              <p className="text-sm font-medium">You're all caught up</p>
   390→              <p className="mt-0.5 text-xs text-muted-foreground">New notifications will appear here.</p>
   391→            </div>
   392→          )}
   393→          {items.map((n, i) => {
   394→            const NIcon = NOTIF_ICONS[n.type] || Bell
   395→            return (
   396→              <button key={n.id} onClick={() => { markOne(n); if (n.link) setModule(n.link) }}
   397→                style={{ ['--delay' as string]: `${Math.min(i * 24, 200)}ms` }}
   398→                className={cn('anim-fade-up flex w-full gap-3 px-4 py-3 text-left transition-colors hover:bg-muted',
   399→                  !n.read && 'bg-primary/[0.04] dark:bg-primary/[0.07]')}>
   400→                <span className={cn('mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg', toneCls[n.type] || toneCls.INFO)}>
   401→                  <NIcon className="h-4 w-4" />

   422→function YearSelector() {
   423→  const { academicYearId, academicYearName, setAcademicYear } = useErp()
   424→  const [years, setYears] = React.useState<any[]>([])

   439→      if (y) setAcademicYear(y.id, y.name)

   441→      <SelectTrigger className="hidden h-9 w-[130px] rounded-xl transition-shadow hover:shadow-sm sm:flex" aria-label="Academic year">

   443→      </SelectTrigger>

   445→        {years.map(y => <SelectItem key={y.id} value={y.id}>{y.name}</SelectItem>)}
   446→      </SelectContent>
   447→    </Select>
   448→  )
